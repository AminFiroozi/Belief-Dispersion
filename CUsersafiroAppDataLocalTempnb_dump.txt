def fetch_from_yfinance(ticker: str, route: str, start_date, end_date, fetch: bool=True):
    price_col_name = ticker.strip('^')
    if fetch:
        df = yf.download(
            tickers=ticker,
            start=start_date,
            end=end_date,
            auto_adjust=False
        )
        
        # 1. Flatten the MultiIndex Columns (Robustness Check)
        if isinstance(df.columns, pd.MultiIndex):
            # If MultiIndex, use the top level names (e.g., 'Adj Close', 'Volume')
            df.columns = df.columns.get_level_values(0)
            
        # 2. Convert the Date Index into a regular column
        df = df.reset_index()
        df['Date'] = pd.to_datetime(df['Date'])
        
        # 3. Check for the price column ('Adj Close' is preferred)
        if 'Adj Close' in df.columns:
            source_col = 'Adj Close'
        elif 'Close' in df.columns:
            source_col = 'Close'
        else:
            raise KeyError(f"Could not find price column ('Adj Close' or 'Close') in yfinance output for {ticker}.")

        # 4. Clean and Select Final Columns (INCLUDING VOLUME)
        
        # The list of columns to keep: Date, the selected price column, and Volume.
        columns_to_keep = ['Date', source_col, 'Volume']
        
        df_final = df[columns_to_keep].copy()
        
        # 5. Rename the source price column to the clean ticker name (e.g., 'SPX')
        df_final.rename(columns={source_col: price_col_name}, inplace=True)
            
        # 6. Save and return
        print(f"Saving cleaned data to {route}")
        df_final.to_csv(route, index=False)
        return df_final
    else:
        df = pd.read_csv(route)
    return df

def load_stored_results():
    """
    Reads the .jsonl file and returns a set of processed INDICES.
    This allows us to skip specific rows we've already done.
    """
    processed_indices = set()
    if os.path.exists(RESULTS_STORAGE_FILE):
        print(f"📂 Reading existing progress from {RESULTS_STORAGE_FILE}...")
        with open(RESULTS_STORAGE_FILE, 'r') as f:
            for line in f:
                if line.strip():
                    try:
                        record = json.loads(line)
                        # Store the dataframe index so we know exactly which row is done
                        processed_indices.add(record['index'])
                    except json.JSONDecodeError:
                        continue
        print(f"✅ Found {len(processed_indices)} previously processed headlines.")
    return processed_indices

def save_batch_results(results_list):
    """
    Appends a list of results to the JSONL file immediately.
    """
    with open(RESULTS_STORAGE_FILE, 'a') as f:
        for record in results_list:
            # Add timestamp for audit
            record['timestamp'] = pd.Timestamp.now().isoformat()
            f.write(json.dumps(record) + '\n')

def get_sentiment_score_finbert(text, tokenizer, model):
    inputs = tokenizer(text, return_tensors="pt", truncation=True, padding=True)
    with torch.no_grad():
        logits = model(**inputs).logits

    probs = F.softmax(logits, dim=-1).squeeze()

    labels = model.config.id2label
    label_scores = {labels[i]: float(probs[i]) for i in range(len(probs))}

    positivity = label_scores.get("positive", 0.0)
    negativity = label_scores.get("negative", 0.0)
    neutrality = label_scores.get("neutral", 0.0)

    sentiment = positivity - negativity

    label = max(label_scores, key=label_scores.get)

    return {
        'sentiment': sentiment,
        'label': label,          # Predicted label (positive, negative, or neutral)
        'positivity': positivity,
        'negativity': negativity,
        'neutrality': neutrality,
    }

current_key_idx = 0
gemini_model_name_idx = 0
openai_api_key = os.getenv("OPENAI_API_KEY")
gemini_api_keys = os.getenv("GOOGLE_API_KEYS")
api_keys = gemini_api_keys.split(",")          # Stored in "A,B,C,D,E" format

def get_response(prompt, LLM, is_json: bool = True):
    try:
        if LLM == "GEMINI":
            gemini_model_name = GEMINI_MODEL_NAMES[gemini_model_name_idx] if CIRCULAR_MODEL else GEMINI_MODEL_NAME
            genai.configure(api_key=api_keys[current_key_idx])
            model = genai.GenerativeModel(
                gemini_model_name,
                generation_config={"response_mime_type": "application/json", "temperature": 0.0}
            )
            response = model.generate_content(prompt)
            text = response.text.strip()
            
        elif LLM == "OPENAI":
            client = OpenAI(api_key=openai_api_key, base_url=BASE_URL)
            response = client.responses.create(
                model=OPENAI_MODEL_NAME,
                input=prompt,
                temperature=0.0
            )
            text = response.output[0].content[0].text
            
        elif LLM == "GROQ":
            client = Groq()
            response = client.chat.completions.create(
                messages=[
                    {
                        "role": "user",
                        "content": prompt,
                    }
                ],
                temperature=0.0,
                model=GROQ_MODEL_NAME,
                max_completion_tokens=65536
            )
            text = response.choices[0].message.content
            
        
        text = re.sub(r"^```json", "", text)
        text = re.sub(r"^```", "", text)
        text = re.sub(r"```$", "", text)
        text = text.strip()
        if is_json:
            try:
                return json.loads(text)
            except json.JSONDecodeError:
                try:
                    return json_repair.loads(text)
                except Exception:
                    raise ValueError(f"Invalid JSON returned:\n{text}")
        else:
           return text
    
    except Exception as e:
        print(f"Error when getting response from LLM, key: {current_key_idx}, model: {GEMINI_MODEL_NAMES[gemini_model_name_idx] if CIRCULAR_MODEL else GEMINI_MODEL_NAME}")
        return None
    

def get_gemini_batch_response(headlines_dict, date_key):
    global current_key_idx
    global gemini_model_name_idx
    
    # Generate prompt with the ID dictionary
    prompt = make_prompt(headlines_dict, date=date_key, mode='collective')
    
    retries = 0
    delay = 2

    while retries <= MAX_API_RETRIES:
        response = get_response(prompt, LLM, True)
        if response is None:
            # Rotate Key
            current_key_idx += 1
            if current_key_idx >= len(api_keys):
                current_key_idx %= len(api_keys)
                gemini_model_name_idx = (gemini_model_name_idx + 1) % len(GEMINI_MODEL_NAMES)
            
            print(f"🔴 Switching to: (Key {current_key_idx}, model {GEMINI_MODEL_NAMES[gemini_model_name_idx]})")
            
            time.sleep(delay)
            delay *= 2
            retries += 1
        else:
            return response
            
    return None

def compute_complex_disparity_features(group, score_col='sentiment_score'):
    """Calculates Gini and Shannon Entropy for a daily group of sentiment scores."""
    
    scores = group[score_col].values
    scores_2d = scores.reshape(-1, 1)

    # Use a slightly less strict sparsity check since basic moments are calculated separately
    if len(scores) < 5: 
        return pd.Series({'gini': np.nan, 'entropy': np.nan})

    # 1. Gini coefficient
    scores_abs = np.sort(np.abs(scores))
    n = len(scores_abs)
    sum_abs = np.sum(scores_abs)
    gini = 0.0
    if sum_abs > 1e-6:
        # Standard Gini formula
        # G = (1 / n) * (1 / mean(y)) * sum[ i * y_i ] - (n+1)/n
        # Simplified: (2 * sum[ i * y_i ] / (n * sum(y))) - (n+1)/n 
        # Using the standard implementation from your original code:
        gini = (np.sum([(2 * i + 1 - n) * scores_abs[i] for i in range(n)])) / (n * sum_abs)

    # 2. Shannon entropy (bin into 10 bins)
    try:
        # Note: Set random_state for reproducibility if used outside of notebook
        binner = KBinsDiscretizer(n_bins=10, encode='ordinal', strategy='uniform', subsample=None) 
        binned = binner.fit_transform(scores_2d).flatten()
        # Calculate the probability distribution (P_i) for the bins
        hist, _ = np.histogram(binned, bins=10, density=True)
        # Calculate Shannon Entropy: H = -sum(P_i * log(P_i))
        entropy_val = entropy(hist + 1e-10) # Avoid log(0)
    except ValueError:
        # This catches "Data is constant" errors
        entropy_val = 0.0
    
    return pd.Series({
        'gini': gini, 
        'entropy': entropy_val
    })

def run_sentiment_regression(
    df: pd.DataFrame, 
    target_col: str, 
    centrality_cols: list, 
    distribution_cols: list, 
    control_cols: list, 
    model_type: str = 'OLS'
):    
    # 1. PREPARE DATA & CREATE LAGS
    # Work on a copy to avoid modifying the original dataframe outside the function
    data = df.copy()
    
    # --- 1.A: AUTOMATIC TARGET EXCLUSION (The Key Improvement) ---
    # Ensure the target variable is not used as a predictor in any feature list.
    def filter_target(target, feature_list):
        return [col for col in feature_list if col != target]

    centrality_cols = filter_target(target_col, centrality_cols)
    distribution_cols = filter_target(target_col, distribution_cols)
    control_cols = filter_target(target_col, control_cols)
    
    # Check if target exists in the DataFrame
    if target_col not in data.columns:
        raise ValueError(f"Target column '{target_col}' not found in DataFrame.")
    
    # Define all predictors after filtering
    all_predictors = centrality_cols + distribution_cols + control_cols
    
    # Create Lags (t-1)
    lagged_centrality = []
    lagged_distribution = []
    lagged_controls = []
    
    for col in all_predictors:
        if col in data.columns:
            lag_name = f"{col}_lag"
            data[lag_name] = data[col].shift(1)
            
            # Categorize the new lag name into the correct list
            if col in centrality_cols:
                lagged_centrality.append(lag_name)
            elif col in distribution_cols:
                # Collinearity check: Exclude std_dev if variance is present
                # Note: This check uses the original, *unlagged* list for definition consistency
                if col == 'std_dev' and 'variance' in distribution_cols:
                    continue
                lagged_distribution.append(lag_name)
            elif col in control_cols:
                lagged_controls.append(lag_name)

    # Drop NaNs created by lagging
    # Dropping from the copy ('data') ensures the original DataFrame ('df') is untouched
    data.dropna(inplace=True)
    
    # --- 1.B: Define Feature Sets (using the lagged, filtered lists) ---
    # 1. Base: Centrality + Controls
    features_base = lagged_centrality + lagged_controls
    # 2. Full: Centrality + Distribution + Controls
    features_full = lagged_centrality + lagged_distribution + lagged_controls
    # 3. Distribution Only
    features_dist = lagged_distribution
    # 4. Controls Only
    features_ctrl = lagged_controls
    
    features_cent = lagged_centrality
    # 5. Distribution + Controls (Added for completeness in some research designs)
    features_dist_ctrl = lagged_distribution + lagged_controls


    # Define Y
    y = data[target_col]

    # Dictionary to store results
    results = {}

    # Helper to run model
    def fit_model(X_names, name):
        """Fits a model, adds a constant, and handles Logit convergence display."""
        if not X_names:
            print(f"⚠️ Warning: No features found for {name} model.")
            return None
            
        # Add intercept (constant) to the design matrix
        X = sm.add_constant(data[X_names], has_constant='add')
        
        # Check for model type
        model_type_upper = model_type.upper()
        if model_type_upper == 'OLS':
            model = sm.OLS(y, X, missing='drop').fit()
        elif model_type_upper == 'LOGIT':
            # disp=0 suppresses convergence messages, good for batch runs
            model = sm.Logit(y, X, missing='drop').fit(disp=0) 
        else:
            raise ValueError("model_type must be 'OLS' or 'Logit'")
            
        return model

    # Print header
    print(f"\n{'='*80}")
    print(f"RUNNING {model_type.upper()} REGRESSION ON TARGET: {target_col}")
    print(f"Number of observations (N): {len(y)}")
    print(f"Predictors Used: {', '.join(all_predictors)}")
    print(f"{'='*80}")

    # 2. RUN MODELS
    models = {
        "Centrality Only": fit_model(features_cent, "Centrality"),
        "Dist Only (Belief Dispersion)": fit_model(features_dist, "Distribution"),
        "Controls Only": fit_model(features_ctrl, "Control"),
        "Dist + Controls": fit_model(features_dist_ctrl, "Dist + Controls"),
        "Base (Centrality+Ctrl)": fit_model(features_base, "Base"),
        "Full (All Features)": fit_model(features_full, "Full"),
    }
    
    # 3. PRINT SUMMARIES & COMPARISONS
    print(f"\n--- Model Performance Summary [{target_col}] ---")
    for name, model in models.items():
        if model:
            results[name] = model
            
            # Extract relevant stats
            R2 = model.prsquared if model_type.upper() == 'LOGIT' else model.rsquared_adj
            AIC = model.aic
            
            # Use f-string for aligned output
            print(f"- {name:<30} | R2/PseudoR2: {R2:.4f} | AIC: {AIC:.2f}")

    # 4. STATISTICAL COMPARISON (Test for Incremental Power)
    base_m = models.get("Base (Centrality+Ctrl)")
    full_m = models.get("Full (All Features)")

    if base_m and full_m:
        print("\n" + "~"*40)
        print("~ MODEL COMPARISON: Full vs Base (Testing Dispersion) ~")
        print("~"*40)
        
        if model_type.upper() == 'OLS':
            # ANOVA F-test (Test for whether the Distribution variables significantly add to the model)
            if full_m.df_resid < base_m.df_resid: # Check if Full is the larger, nested model
                # Note: statsmodels requires the smaller model first, then the larger (Full)
                anova = anova_lm(base_m, full_m)
                print("ANOVA F-Test (H0: Distribution Coeffs = 0):")
                print(anova)
                results['anova'] = anova
            else:
                 print("ANOVA skipped: Full model must contain all Base model variables.")
        
        else: # LOGIT
            # Likelihood Ratio Test (LR Test)
            lr_stat = 2 * (full_m.llf - base_m.llf)
            # df difference is the number of added predictors (the Dispersion variables)
            df_diff = full_m.df_model - base_m.df_model
            p_val = chi2.sf(lr_stat, df_diff)
            
            print(f"Likelihood Ratio Stat (Chi2): {lr_stat:.4f}")
            print(f"Degrees of Freedom: {df_diff}")
            print(f"P-value: {p_val:.4e} (rejects H0 if p < alpha)")
            
            # Include in results for programatic access
            results['lr_test'] = {
                'stat': lr_stat, 
                'p_value': p_val, 
                'df': df_diff
            }
            
    # Include a flag for the target column used
    results['target_col'] = target_col

    return results

coefs = model_full_vol.params.drop('const', errors='ignore').rename('Coefficient')
conf_int = model_full_vol.conf_int().drop('const', errors='ignore')
conf_int.columns = ['Lower CI', 'Upper CI']

# Align and combine data
plot_data = pd.concat([coefs, conf_int], axis=1).sort_values(by='Coefficient', ascending=True)

# Create human-readable labels
labels = {
    'soft_mean_lag': 'Soft Mean (Centrality)',
    'return_lag': 'Lagged Return (Control)',
    'volume_lag': 'Lagged Volume (Control)',
    'vix_lag': 'Lagged VIX (Control)',
    'variance_lag': 'Variance (Distribution)',
    'skewness_lag': 'Skewness (Distribution)',
    'kurtosis_lag': 'Kurtosis (Distribution)',
    'gini_lag': 'Gini (Distribution)',
    'entropy_lag': 'Entropy (Distribution)',
}
plot_data.index = [labels.get(i, i.replace('_lag', '').capitalize()) for i in plot_data.index]

plt.figure(figsize=(10, 8))
# Plot coefficients (points) and confidence intervals (lines)
plt.errorbar(plot_data['Coefficient'], plot_data.index, 
             xerr=[plot_data['Coefficient'] - plot_data['Lower CI'], 
                   plot_data['Upper CI'] - plot_data['Coefficient']],
             fmt='o', color='royalblue', capsize=4, ecolor='lightcoral', elinewidth=1, markerfacecolor='royalblue')
plt.axvline(0, color='grey', linestyle='--', linewidth=0.8)
plt.title('Full OLS Model: Coefficients and 95% C.I. for Volatility Prediction', fontsize=14)
plt.xlabel('Coefficient Value', fontsize=12)
plt.ylabel('Predictor Variable', fontsize=12)
plt.grid(axis='x', linestyle=':', alpha=0.6)
plt.tight_layout()
plt.savefig('ols_full_model_coefficients.png')
plt.close()
